## 🤖 3. Regra de Ativação Automática no Python (`scripts/gerador_filmeja.py`)

def selecionar_autor(genero: str, tema: str) -> dict:
    texto = f"{genero} {tema}".lower()
    
    # 1. Séries, episódios e lançamentos semanais -> Marcos Vinicius
    if any(k in texto for k in ["serie", "série", "episodio", "episódio", "temporada", "quando sai", "que dia sai", "materia escura", "apple tv"]):
        return AUTHORS["marcos"]
        
    # 2. Ficção Científica & Espaço -> Camila Fontes
    if any(k in texto for k in ["sci-fi", "ficcao", "ficção", "espaco", "espaço", "alien", "duna"]):
        return AUTHORS["camila"]
        
    # 3. Ação & Sobrevivência -> Thiago Rocha
    if any(k in texto for k in ["acao", "ação", "sobrevivencia", "sobrevivência", "adrenalina"]):
        return AUTHORS["thiago"]
        
    # 4. Policial, Mistério & Fincher -> Guilherme Santos
    if any(k in texto for k in ["policial", "misterio", "mistério", "investigacao", "crime", "whodunit"]):
        return AUTHORS["guilherme"]
        
    # 5. Slasher, Sobrenatural & Demônios -> Beatriz Silveira
    if any(k in texto for k in ["slasher", "sobrenatural", "assombrada", "demonio", "body horror"]):
        return AUTHORS["beatriz"]
        
    # 6. Horror de Autor e Filmes Premiados -> Lucas Andrade (Padrão)
    return AUTHORS["lucas"]
